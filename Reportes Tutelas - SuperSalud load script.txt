///$tab 14. Union
// Subrutina para cargar un QVD, renombrar sus campos con prefijo único y crear el campo Registro completo
SUB CargarYUnirQVD(vArchivo, vTablaDestino, vFuente)
    // 1. Cargar campos y renombrarlos con prefijo
    TempTable:
    LOAD
        // Creamos dinámicamente nombres únicos: Prefijo_Campo
        // Iteramos con la función FieldName() para obtener cada nombre y agregarle el prefijo
        // La macro Replace() quita espacios para evitar errores
        // Nota: No se puede usar directamente FieldName() en el script, por eso el renombrado se hace en la siguiente etapa
        * 
    FROM [$(vArchivo)] (qvd);

    // 2. Renombrar los campos con el prefijo de la fuente
    FOR i = 1 TO NoOfFields('TempTable')
        LET vOldName = FieldName($(i), 'TempTable');
        LET vNewName = Replace('$(vFuente)&"_"&$(vOldName)', ' ', '_'); // Prefijo + Campo (sin espacios)
        RENAME FIELD [$(vOldName)] TO [$(vNewName)];
    NEXT

    // 3. Crear la cadena concatenada dinámica
    LET vNumFields = NoOfFields('TempTable');
    LET vConcat = '';

    FOR i = 1 TO $(vNumFields)
        LET vFieldName = FieldName($(i), 'TempTable');
        IF $(i) = $(vNumFields) THEN
            LET vConcat = '$(vConcat) [' & '$(vFieldName)' & ']';
        ELSE
            LET vConcat = '$(vConcat) [' & '$(vFieldName)' & '] & CHR(124) &';
        ENDIF
    NEXT

    // 4. Cargar a la tabla final con el campo Fuente y el Registro completo
    $(vTablaDestino):
    LOAD
        *,
        '$(vFuente)' AS Fuente,
        $(vConcat) AS "Registro completo"
    RESIDENT TempTable;

    // 5. Limpiar tabla temporal
    DROP TABLE TempTable;
END SUB
